select
cj.业务组
,cj.城市
,city.城市经理
,cj.战区
,zq.战区经理
,sum(cj.成交额) 成交总和
,sum(hk.注册人数) 注册人数总和
,sum(cj.逾期金额) 逾期总和
,round(sum(cj.逾期金额)/sum(cj.成交额),4) 逾期率
from dsx_test.`8月成交数据` cj
join dsx_test.`8月获客数据` hk
on cj.日期 = hk.日期 and cj.业务组 = hk.业务组
join dsx_test.城市信息 city
on cj.城市 = city.城市名称
join dsx_test.战区信息 zq
on cj.战区 = zq.战区名称
where cj.成交额 >  100000 and hk.注册人数 > 5000
group by  1
having sum(逾期金额) < 100000

计算战区下城市平均成交额大于5百万的战区的总成交额
select sum(成交额) 满足条件战区总成交额
from dsx_test.`8月成交数据` a1
join
(
    select 战区
from dsx_test.`8月成交数据`
group by 1
having sum(成交额)/count(distinct 城市) > 5000000
) a2
on a1.战区 = a2.战区
求出成交额小于城市平均成交额的城市在各自战区的总成交额，按成交额降序排序
select 战区
,sum(成交额) 满足条件城市聚合到战区成交额
from dsx_test.`8月成交数据` a1
right join
(
    select 城市
    from dsx_test.`8月成交数据`
    group by 1
    having sum(成交额) < (select sum(成交额)/count(distinct 城市) from dsx_test.8月成交数据)
) a2
on a2.城市 = a1.城市
group by 1
order by 2 desc
求8月各战区成交额最高的三天、这三天的排行、成交额、战区总成交额、以及各自占战区总成交额的占比（保留小数点后3位小数）
select *
,round(成交额/战区总成交额,3) 总成交额占比
from (select 战区
           , 日期
           , 成交额
           , rank() over (partition by 战区 order by 成交额 desc) 成交额排行
           , sum(成交额) over (partition by 战区)                 战区总成交额
      from (select 战区
                 , cast(replace(replace(replace(日期, '年', '-'), '月', '-'), '日', '-') as date) 日期
                 , sum(成交额) 成交额
            from dsx_test.`8月成交数据`
            group by 1, 2
            )a
      order by 1,4
    )b
where 成交额排行 < 4
求8月各战区日成交额排名前20%的日期、成交额、以及具体排名及排名百分比（以百分比展示，取百分比后一位小数）
select
    战区
    ,日期
    ,成交额
    ,成交额排行
    ,concat(round(排名百分比*100,1),'%') 排名百分比
from(
            select
            战区
            ,日期
            ,成交额
            ,rank() over (partition by 战区 order by 成交额 desc) 成交额排行
            ,cume_dist() over (partition by 战区 order by 成交额 desc) 排名百分比
            from (
                 select
                 战区
                ,日期
                ,sum(成交额) 成交额
                 from dsx_test.`8月成交数据`
                 group by 1,2
                ) a
        ) b
where 排名百分比 < 0.2
求8月各战区日成交额的极差，按极差降序排序 | 解法一：窗口函数
select 战区
,(最大值-最小值) 极差
from
    (
        select 战区
        ,日期
        ,成交额
        ,max(成交额) over (partition by 战区) 最大值
        ,min(成交额) over (partition by 战区) 最小值
        from(
            select 战区
            ,日期
            ,sum(成交额) 成交额
            from dsx_test.`8月成交数据`
            group by 1,2
            ) a
    ) b
group by 1
order by 2 desc

求8月东部战区的日成交额，天环比，周同比
select
    战区
    ,日期
    ,成交额
    ,concat(round((成交额/前一日成交额-1)*100,1),'%') 天环比
    ,concat(round((成交额/前一周成交额-1)*100,1),'%') 周同比
from
    (
        select
            战区
            ,日期
            ,成交额
            ,lag(成交额,1) over (partition by 战区 order by 日期) 前一日成交额
            ,lag(成交额,7) over (partition by 战区 order by 日期) 前一周成交额
        from (
            select
                战区
                ,cast(replace(replace(replace(日期,'年','-'),'月','-'),'日','-') as date) 日期
                ,sum(成交额) 成交额
                from dsx_test.`8月成交数据`
                where 战区 = '东部战区'
                group by 1,2
             ) a
    ) b
求8月东部战区的日成交额、当月总成交额、截至当天的累计成交额、近3天的平均成交额（当日及前2日）、近7天的累计成交额（当日及前后3日）
select
     战区
     ,日期
     ,成交额 日成交额
     ,sum(成交额) over (partition by 战区) 当月总成交额
     ,sum(成交额) over (partition by 战区 order by 日期) 截止当天的累计成交额
     ,avg(成交额) over (partition by 战区 order by 日期 rows 2 preceding) 近三天的平均成交额
     ,sum(成交额) over (partition by 战区 order by 日期 rows between 3 preceding and 3 following) 近七天的累计成交额
from (
        select
        战区
        ,cast(replace(replace(replace(日期,'年','-'),'月','-'),'日','-') as date) 日期
        ,sum(成交额) 成交额
        from dsx_test.`8月成交数据`
        where 战区 = '东部战区'
        group by 1,2
) a
